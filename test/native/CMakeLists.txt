cmake_minimum_required(VERSION 3.22)
project(wheelplay_media_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT LIBDATACHANNEL_SOURCE_DIR)
    message(FATAL_ERROR "Set LIBDATACHANNEL_SOURCE_DIR to the pinned libdatachannel checkout")
endif()
set(NO_TESTS ON CACHE BOOL "" FORCE)
set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
set(NO_WEBSOCKET ON CACHE BOOL "" FORCE)
set(NO_MEDIA OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(libdatachannel_SOURCE_DIR "${LIBDATACHANNEL_SOURCE_DIR}")
add_subdirectory("${LIBDATACHANNEL_SOURCE_DIR}" rtc)
include(../../common/src/main/cpp/media_send_feedback.cmake)
include(../../common/src/main/cpp/srtp_optimization.cmake)
enable_testing()
foreach(name media-send video-packetizer video-rtp-handler srtp-packet)
    add_executable(${name} ${name}.cpp)
    target_include_directories(${name} PRIVATE ../../common/src/main/cpp)
    target_compile_options(${name} PRIVATE -UNDEBUG)
    add_test(NAME ${name} COMMAND ${name})
endforeach()
target_link_libraries(video-rtp-handler PRIVATE datachannel-static srtp2)
target_link_libraries(srtp-packet PRIVATE srtp2)
